"""
$description - Action steps

Action stage: fetch data with spider (PlaywrightSpider).
"""

from auto_spider import action, Context


@action()
def fetch_page(context: Context):
    """Navigate to URL and get page info with interactive elements."""
    url = context.task.get('url')
    if not url:
        raise ValueError("Task must have 'url' field")
    
    # navigate to page
    content = context.spider.do_url(url)
    
    # get page info with interactive elements via CDP
    page_info = context.spider.get_page_info()

    # save raw HTML to context['content']
    context['content'] = content
    
    # save page info to context['result']
    context['result'] = page_info
    
    # action stage saves:
    # - task1_task.json (original task from context.task)
    # - task1_action.json (page_info from context['result'])
    # - task1.html (content from context['content'])
    
    return content
