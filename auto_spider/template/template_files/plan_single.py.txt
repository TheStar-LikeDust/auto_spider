@action()
def fetch_page(context: Context):
    """Navigate to URL and get page info with interactive elements."""
    url = context.task.get('url')
    if not url:
        raise ValueError("Task must have 'url' field")

    content = context.spider.do_url(url)
    markdown = html_to_markdown(content)
    cleaned_html = clean_html(content).get('body', '')
    page_info = context.spider.get_page_info()

    context['content'] = content
    context['result'] = {
        'markdown': markdown,
        'cleaned_html': cleaned_html,
        'page_info': page_info,
    }
    return content
